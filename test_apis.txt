# 测试脚本
# 在终端中逐条运行

# 1. 先登录拿到 token
curl -s -X POST http://localhost:5228/auth/login \
  -H "Content-Type: application/json" \
  -d '{"login_id":"cc专用","password":"abc123456"}'

# 把上面返回的 access_token 粘贴到下面
TOKEN="这里填token"

# 2. 测试工具自动发现
echo "=== 工具总数 ==="
curl -s http://localhost:5228/admin/tools -H "Authorization: Bearer $TOKEN" | python3 -c "import sys,json; d=json.load(sys.stdin); print(f'工具总数: {d[\"total\"]}')"

# 3. 测试插件管理器
echo "=== 插件列表 ==="
curl -s http://localhost:5228/admin/plugins -H "Authorization: Bearer $TOKEN" | python3 -c "import sys,json; [print(f'  {p[\"name\"]}: running={p[\"running\"]}') for p in json.load(sys.stdin)['plugins']]"

# 4. 测试 browser-status 向后兼容
echo "=== browser-status ==="
curl -s http://localhost:5228/admin/browser-status -H "Authorization: Bearer $TOKEN"

# 5. 查看后端启动日志
echo "=== 后端启动日志（最后20行）==="
docker compose logs --tail=20 backend
